\begin{frame}{The recruitment gap}

{\footnotesize rise in that share\\since 2022\\(64\% $\rightarrow$ 86\%)}
\end{column}

\begin{pbox}
\small

\end{pbox}
\src{Figures as supplied by the team; cite original sources before submission. 22 pts = 86 $-$ 64.}

% ---- 03 PITCH ----
\begin{frame}{One-slide pitch}
\begin{tbox}{> rematch --explain}
\small
\begin{enumerate}
\item Patient connects records once (simulated Epic)
\item Deterministic matching finds what may fit
\item Doctor only when risk needs a conversation
\item Patient authorizes each introduction --- exact packet
\end{enumerate}
\end{tbox}
\vskip10pt
\centering
\inv{Screening support}\quad not eligibility\quad //\quad
\inv{Patient chooses}\quad not clinic push
\end{frame}

% ---- 05 DEMO: SETUP (app live) ----
\begin{frame}{Live: study setup → PCP path}
\begin{columns}[T]
\begin{column}{0.32\textwidth}
\begin{tbox}{Happening}
\small
Team lists a higher-risk study. Rules + brief from public text. Route = \alert{HIGH}.
\end{tbox}
\end{column}
\begin{column}{0.32\textwidth}
\begin{tbox}{Why helpful}
\small
Matching runs without pulling the PCP into every import.
\end{tbox}
\end{column}
\begin{column}{0.32\textwidth}
\begin{tbox}{Resistance gone}
\small
No clinic invitation gate. No pre-auth candidate list for the team.
\end{tbox}
\end{column}
\end{columns}
\vskip12pt
\centering\small\texttt{\$ switch → Study team · list SYN-HIGH}\,\terminalcursor
\end{frame}

% ---- 09 ARCHITECTURE ----
\begin{frame}{Information flow \& safeguards}
\begin{columns}[T]
\begin{column}{0.52\textwidth}
\begin{tbox}{> flow}
\footnotesize
\begin{enumerate}
\item Public study text $\rightarrow$ LLM rules
\item Patient grants matching + records
\item Deterministic match (no LLM on PHI)
\item LOW $\rightarrow$ patient; HIGH/UNK $\rightarrow$ PCP
\item Exact preview $\rightarrow$ patient auth
\item Named team only
\end{enumerate}
\end{tbox}
\end{column}
\begin{column}{0.45\textwidth}
\begin{tbox}{> never}
\footnotesize
\begin{itemize}
\item PHI in model prompts
\item Team browsing candidates
\item Full chart by default
\item Eligibility by ReMatch
\end{itemize}
\end{tbox}
\vskip8pt
\begin{pbox}
\footnotesize
AES-GCM at rest · versioned permissions · withdraw / delete
\end{pbox}
\end{column}
\end{columns}
